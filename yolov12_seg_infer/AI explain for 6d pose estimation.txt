class CameraSystem(Node):

    def __init__(self):

        super().__init__('camera_system_sub')

        # Verify these exactly match your 'ros2 topic list'

        self.create_subscription(Image, '/rgbd_camera/image', self.rgb_cb, 10)

        self.create_subscription(Image, '/rgbd_camera/depth_image', self.depth_cb, 10)

        self.create_subscription(Bool, '/trigger', self.trigger_cb, 10)


    def rgb_cb(self, msg):

        global rgb_image

        try:

            # Gazebo publishes rgb8, convert to bgr8 for OpenCV

            rgb_image = bridge.imgmsg_to_cv2(msg, "bgr8")

            self.get_logger().info("RGB image received!", once=True)

        except Exception as e:

            self.get_logger().error(f"RGB conversion error: {e}")


    def depth_cb(self, msg):

        global depth_image

        try:

            # GZ Sim often sends 32FC1 (float32)

            depth_image = bridge.imgmsg_to_cv2(msg, 'passthrough')

            self.get_logger().info("Depth image received!", once=True)

        except Exception as e:

            self.get_logger().error(f"Depth conversion error: {e}")


    def trigger_cb(self, msg):

        global trigger_signal

        trigger_signal = msg.data

        self.get_logger().info(f"Trigger State: {trigger_signal}")


class NutEstimatorAAE(Node):

    def __init__(self):

        super().__init__('aae_estimator_node')

        # Force YOLO to use CPU only

        self.model = YOLO("/home/hussain/6d_pose_estimation/nut_pose_pipeline/yolov12_seg_infer/model/best.pt")

        self.model.to('cpu')  # Force CPU usage

        self.get_logger().info("YOLO model loaded on CPU")

        

        self.K = np.array([[615.0, 0, 320.0], [0, 615.0, 240.0], [0, 0, 1]], dtype=np.float32)

        self.tf_broadcaster = TransformBroadcaster(self)

        self.tf_buffer = Buffer()

        self.tf_listener = TransformListener(self.tf_buffer, self)

        self.is_tf_ready = False

        self.T_wc = np.eye(4)

        

        self.create_timer(0.1, self.process_frame)

        self.get_logger().info("v3 Node Started - Visualizer Active")


    def draw_3d_axis(self, img, R, t):

        axis_len = 0.05

        axis_pts = np.float32([[0,0,0], [axis_len,0,0], [0,axis_len,0], [0,0,axis_len]]).reshape(-1, 3)

        rvec, _ = cv2.Rodrigues(R)

        imgpts, _ = cv2.projectPoints(axis_pts, rvec, t.reshape(3,1), self.K, None)

        imgpts = imgpts.astype(int)

        origin = tuple(imgpts[0].ravel())

        cv2.line(img, origin, tuple(imgpts[1].ravel()), (0,0,255), 2) # X

        cv2.line(img, origin, tuple(imgpts[2].ravel()), (0,255,0), 2) # Y

        cv2.line(img, origin, tuple(imgpts[3].ravel()), (255,0,0), 2) # Z

        return img


    def process_frame(self):

        global rgb_image, depth_image, trigger_signal

        

        # Check Data Presence

        if rgb_image is None: return

        

        display_img = rgb_image.copy()


        # Check Trigger

        if not trigger_signal:

            cv2.putText(display_img, "STATUS: IDLE (WAITING FOR TRIGGER)", (10, 30), 

                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)

            cv2.imshow("6D Pose Visualization", display_img)

            cv2.waitKey(1)

            return


        # Check Depth

        if depth_image is None:

            cv2.putText(display_img, "STATUS: MISSING DEPTH DATA", (10, 30), 

                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 255), 2)

            cv2.imshow("6D Pose Visualization", display_img)

            cv2.waitKey(1)

            self.get_logger().info(f"MISSING DEPTH DATA")

            


        # TF Handling

        if not self.is_tf_ready:

            try:

                t = self.tf_buffer.lookup_transform('world', 'camera_link_optical', rclpy.time.Time())

                self.T_wc = self.transform_to_matrix(t)

                self.is_tf_ready = True

                self.get_logger().info(f"TF linked")

            except:

                return


        # Inference Logic (now on CPU)

        self.get_logger().info(f"model will be load")

        results = self.model(rgb_image, conf=0.6, verbose=False, device='cpu')

        for r in results:

            if r.masks is not None:

                display_img = r.plot()

                mask = r.masks.data[0].cpu().numpy()

                x1, y1, x2, y2 = map(int, r.boxes.xyxy[0].cpu().numpy())

                crop = rgb_image[y1:y2, x1:x2]

                self.get_logger().info(f"model is loaded")

                

                try:

                    _, enc = cv2.imencode('.jpg', crop)

                    resp = requests.post('http://localhost:5001/get_rotation', files={"image": enc.tobytes()}, timeout=30.0)

                    R_mat = np.array(resp.json()['R'])

                    

                    # Depth Calculation

                    z_m = np.median(depth_image[mask > 0])

                    if np.isnan(z_m) or z_m <= 0: continue

                    if z_m > 10: z_m /= 1000.0 # Convert mm to m

                    

                    cx, cy = r.boxes.xywh[0][0].cpu().item(), r.boxes.xywh[0][1].cpu().item()

                    tx = (cx - self.K[0, 2]) * z_m / self.K[0, 0]

                    ty = (cy - self.K[1, 2]) * z_m / self.K[1, 1]

                    T_vec = np.array([tx, ty, z_m])


                    display_img = self.draw_3d_axis(display_img, R_mat, T_vec)

                    self.broadcast_pose(self.T_wc @ self.compose_mat(R_mat, T_vec))

                except Exception as e:

                    self.get_logger().error(f"Inference Step Error: {e}")


        cv2.imshow("6D Pose Visualization", display_img)

        cv2.waitKey(1)


    def transform_to_matrix(self, t):

        m = np.eye(4)

        m[:3, 3] = [t.transform.translation.x, t.transform.translation.y, t.transform.translation.z]

        q = [t.transform.rotation.x, t.transform.rotation.y, t.transform.rotation.z, t.transform.rotation.w]

        m[:3, :3] = R_tool.from_quat(q).as_matrix()

        return m


    def compose_mat(self, R, t):

        m = np.eye(4); m[:3, :3], m[:3, 3] = R, t

        return m


    def broadcast_pose(self, mat):

        now = self.get_clock().now().to_msg()

        t = TransformStamped()

        

        t.header.stamp = now

        t.header.frame_id = 'world'

        t.child_frame_id = 'detected_nut'

        

        t.transform.translation.x = float(mat[0, 3])

        t.transform.translation.y = float(mat[1, 3])

        t.transform.translation.z = float(mat[2, 3])

        

        q = R_tool.from_matrix(mat[:3, :3]).as_quat()

        t.transform.rotation.x = float(q[0])

        t.transform.rotation.y = float(q[1])

        t.transform.rotation.z = float(q[2])

        t.transform.rotation.w = float(q[3])


        try:

            if hasattr(self.tf_broadcaster, 'send_transform'):

                self.tf_broadcaster.send_transform(t)

            elif hasattr(self.tf_broadcaster, 'sendTransform'):

                self.tf_broadcaster.sendTransform(t)

            else:

                self.get_logger().error(f"Broadcaster methods: {dir(self.tf_broadcaster)}")

        except Exception as e:

            self.get_logger().error(f"TF Failed but continuing: {e}")
